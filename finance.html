function generateOfficialInvoice() { const comp = document.getElementById('invoiceTargetCompany').value; if(!comp) { alert("請選擇請款母公司！"); return; } const profile = { name: document.getElementById('ciName').value, taxId: document.getElementById('ciTaxId').value, contact: document.getElementById('ciContact').value, phone: document.getElementById('ciPhone').value, address: document.getElementById('ciAddress').value, terms: document.getElementById('ciTerms').value, method: document.getElementById('ciMethod').value }; let invoiceData = rawLogisticsData; if(invoiceData.length === 0) { alert("無任何出貨明細可產生對帳單！"); return; } const printDate = getLocalDateString(new Date()); const period = `${document.getElementById('startDate').value} ~ ${document.getElementById('endDate').value}`; let deptsData = {}; let deptItemSummary = {}; // ★ 改變資料結構：改為依「部門」分開統計 let grandTotal = 0; invoiceData.forEach(r => { let dName = r.hotel; if(!deptsData[dName]) deptsData[dName] = []; if(!deptItemSummary[dName]) deptItemSummary[dName] = {}; // 初始化該部門的統計物件 let price = r.currentPrice !== undefined ? r.currentPrice : (getPrice(r.hotel, r.item) || 0); let amt = r.qty * price; grandTotal += amt; deptsData[dName].push({ date: r.date, item: r.item, qty: r.qty, price: price, amt: amt, receiptOrder: r.receiptOrder }); // ★ 依「部門 -> 品項」分別加總數量與金額 if(!deptItemSummary[dName][r.item]) deptItemSummary[dName][r.item] = { qty: 0, price: price, amt: 0 }; deptItemSummary[dName][r.item].qty += r.qty; deptItemSummary[dName][r.item].amt += amt; }); let html = `
百麒國際股份有限公司
Bai-qi International CO., LTD
客 戶 對 帳 單
ACCOUNT STATEMENT
製表日期：${printDate}
頁碼：1 / ${Object.keys(deptsData).length + 1}
客戶資訊
客戶名稱${profile.name}
統一編號${profile.taxId}
聯絡人${profile.contact}
聯絡電話${profile.phone}
出貨地址${profile.address}
對帳資訊
對帳期間${period}
幣別新台幣 (NTD)
付款條件${profile.terms}
付款方式${profile.method}
備註如有疑義，請於收到對帳單後 7 日內與本公司聯繫。
`; let idx = 1; for (let [deptName, records] of Object.entries(deptsData)) { let deptSum = records.reduce((sum, r) => sum + r.amt, 0); html += ``; } let tax = Math.round(grandTotal * 0.05); let totalWithTax = grandTotal + tax; html += `
項次部門名稱結算總計 (未稅)
${idx++}${deptName} 洗滌費結算${deptSum.toLocaleString()}
各部門 品項統計明細
`; // ★ 迴圈渲染：將每個部門標題列出，接著列出其下品項 for (let [deptName, items] of Object.entries(deptItemSummary)) { // 部門大標題 html += ``; let deptSubtotal = 0; for (let [itemName, data] of Object.entries(items)) { // 該部門內的各個品項 (稍微縮排) html += ``; deptSubtotal += data.amt; } // 該部門的小計 html += ``; } html += `
品項名稱 總數量 單價 總金額
🏨 ${deptName}
${itemName}${data.qty.toLocaleString()}$${data.price.toLocaleString()}$${data.amt.toLocaleString()}
${deptName} 統計小計$${deptSubtotal.toLocaleString()}
結算總計
未稅金額${grandTotal.toLocaleString()}
稅金 (5%)${tax.toLocaleString()}
含稅金額$${totalWithTax.toLocaleString()}
備註事項：
1. 本對帳單僅供雙方核對交易紀錄，實際金額以本公司稅務憑證為準。
2. 逾期未提出異議者，視同確認本對帳單內容無誤。
`; let pageNum = 2; for (let [deptName, records] of Object.entries(deptsData)) { records.sort((a,b) => { if(a.date !== b.date) return a.date < b.date ? -1 : 1; return a.receiptOrder - b.receiptOrder; }); html += `
【附件】交易明細清單
客戶：${profile.name} ｜ 部門：${deptName}
頁碼：${pageNum++} / ${Object.keys(deptsData).length + 1}
`; let lastDate = ""; let deptSum = 0; records.forEach(r => { let dateDisplay = (r.date !== lastDate) ? r.date : ""; lastDate = r.date; deptSum += r.amt; html += ``; }); html += `
日期品項名稱數量 (飯店實點)單價金額
${dateDisplay}${r.item}${r.qty}${r.price}${r.amt.toLocaleString()}
${deptName} 結算小計：${deptSum.toLocaleString()}
`; } document.getElementById('printArea').innerHTML = html; closeInvoiceModal(); document.querySelector('.app-container').classList.add('hidden'); document.querySelector('.excel-bottom-bar').classList.add('hidden'); document.body.style.background = 'white'; document.getElementById('printControls').classList.replace('hidden', 'flex'); }
